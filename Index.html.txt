        function playUiSound(type = 'click') {
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.connect(gain);
                gain.connect(ctx.destination);
                
                if (type === 'click') {
                    osc.frequency.setValueAtTime(440, ctx.currentTime);
                    gain.gain.setValueAtTime(0.05, ctx.currentTime);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.05);
                } else if (type === 'success') {
                    osc.frequency.setValueAtTime(587.33, ctx.currentTime);
                    gain.gain.setValueAtTime(0.08, ctx.currentTime);
                    osc.start();
                    osc.stop(ctx.currentTime + 0.15);
                }
            } catch(e) {}
        }

        function handleChatSubmit(e) {
            if (e && e.preventDefault) e.preventDefault();
            const input = document.getElementById('chat-input');
            const message = input.value.trim();
            if (!message) return;

            playUiSound('click');
            appendChatMessage(message, 'user');

                appendChatMessage(response, 'bot');
                playUiSound('success');
            }, 600);
        }

        function searchWasteItem(query) {
            const box = document.getElementById('search-result-box');
            if (!query.trim()) {
                box.classList.add('hidden');
                return;
            }
            
            const q = query.toLowerCase();
            let binResult = 'Dry / Plastic Bin';
            if (q.includes('pizza') || q.includes('food') || q.includes('banana') || q.includes('apple')) binResult = 'Green Organic Bin';
            if (q.includes('battery') || q.includes('phone') || q.includes('wire')) binResult = 'Special E-Waste Kiosk';
            if (q.includes('bottle') || q.includes('jar')) binResult = 'Glass / Recycling Bin';

            box.innerHTML = `
                <div class="flex items-center gap-2">
                    <i class="fa-solid fa-circle-info text-emerald-500 text-sm"></i>
                    <div>
                        <strong>"${query}"</strong> should be disposed in: 
                        <span class="font-extrabold text-emerald-600 dark:text-emerald-400 block">${binResult}</span>
                    </div>
                </div>
            `;
            box.classList.remove('hidden');
        }

        /* GIS INTERACTIVE CANVAS MAP */
        let mapZoom = 1;
        function renderGisMap() {
            const canvas = document.getElementById('gis-map-canvas');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');
            
            canvas.width = canvas.parentElement.clientWidth;
            canvas.height = canvas.parentElement.clientHeight;

            ctx.fillStyle = '#0f172a';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            ctx.strokeStyle = '#1e293b';
            ctx.lineWidth = 1;
            for(let x=0; x<canvas.width; x+=30) {
                ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, canvas.height); ctx.stroke();
            }
            for(let y=0; y<canvas.height; y+=30) {
                ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(canvas.width, y); ctx.stroke();
            }

            ctx.strokeStyle = '#334155';
            ctx.lineWidth = 14;
            ctx.beginPath(); ctx.moveTo(20, 100); ctx.lineTo(canvas.width-20, 100); ctx.stroke();
            ctx.beginPath(); ctx.moveTo(150, 10); ctx.lineTo(150, canvas.height-10); ctx.stroke();

            state.bins.forEach(bin => {
                if (state.activeBinFilter !== 'all' && bin.type !== state.activeBinFilter) return;

                const x = bin.lat * mapZoom;
                const y = bin.lng * mapZoom;

                ctx.beginPath();
                ctx.arc(x, y, 14, 0, Math.PI * 2);
                ctx.fillStyle = bin.fill > 80 ? 'rgba(239, 68, 68, 0.3)' : 'rgba(46, 204, 113, 0.3)';
                ctx.fill();

                ctx.beginPath();
                ctx.arc(x, y, 7, 0, Math.PI * 2);
                ctx.fillStyle = bin.fill > 80 ? '#ef4444' : '#2ecc71';
                ctx.fill();
                ctx.strokeStyle = '#ffffff';
                ctx.lineWidth = 2;
                ctx.stroke();

                ctx.fillStyle = '#ffffff';
                ctx.font = 'bold 9px Inter';
                ctx.fillText(`${bin.fill}%`, x - 8, y - 10);
            });

            const container = document.getElementById('bins-list-container');
            if(!container) return;
            container.innerHTML = state.bins.map(bin => `
                <div onclick="selectBin('${bin.id}')" class="bg-white dark:bg-brand-800 p-3 rounded-xl border border-gray-100 dark:border-brand-700/50 flex justify-between items-center cursor-pointer hover:border-brand-500 transition">
                    <div class="flex items-center gap-2.5">
                        <div class="w-8 h-8 rounded-full ${bin.fill > 80 ? 'bg-red-100 text-red-600' : 'bg-emerald-100 text-emerald-600'} flex items-center justify-center text-xs font-bold">
                            <i class="fa-solid fa-trash"></i>
                        </div>
                        <div>
                            <h4 class="font-bold text-xs text-gray-800 dark:text-gray-100">${bin.name}</h4>
                            <span class="text-[10px] text-gray-400">${bin.type} Bin • ${bin.distance}</span>
                        </div>
                    </div>
                    <span class="text-xs font-black ${bin.fill > 80 ? 'text-red-500' : 'text-emerald-500'}">${bin.fill}% Full</span>
                </div>
            `).join('');
        }

        }

            const container = document.getElementById('home-recent-feed');
            if (!container) return;

                <div onclick="openReportModal('${report.id}')" class="bg-white dark:bg-brand-800 p-3 rounded-2xl shadow-sm border border-gray-100 dark:border-brand-700/50 flex gap-3 cursor-pointer hover:border-brand-500 transition">
                    <img src="${report.image}" class="w-16 h-16 rounded-xl object-cover">
                    <div class="flex-1 min-w-0">

                            <span class="text-[10px] font-bold text-brand-500">#${report.id}</span>

                        </div>
                        <h4 class="font-bold text-xs text-gray-800 dark:text-gray-100 truncate mt-0.5">${report.category}</h4>
                        <p class="text-[11px] text-gray-400 truncate">${report.area}</p>

        function getStatusBadgeClass(status) {
            if (status === 'Resolved') return 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300';
            if (status === 'In Progress') return 'bg-blue-100 text-blue-700 dark:bg-blue-900/50 dark:text-blue-300';
            if (status === 'Under Review') return 'bg-purple-100 text-purple-700 dark:bg-purple-900/50 dark:text-purple-300';
            return 'bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-300';
        }

        function filterCommunityFeed(filter) {
            playUiSound('click');
            state.activeReportFilter = filter;
            renderCommunityFeed();
        }

        function upvoteReport(id) {
            playUiSound('click');
            const report = state.reports.find(r => r.id === id);
            if (report) {
                report.upvotes += 1;
                renderCommunityFeed();
                showToast('Cleanup prioritized (+1 Upvote)');
            }
        }

        /* REPORT DETAIL MODAL */
        function openReportModal(reportId) {
            playUiSound('click');
            state.selectedReportId = reportId;
            const report = state.reports.find(r => r.id === reportId);
            if (!report) return;

            document.getElementById('modal-report-id').textContent = `#${report.id}`;
            document.getElementById('modal-report-title').textContent = report.category;
            document.getElementById('modal-status-text').textContent = report.status;
            document.getElementById('modal-report-img').src = report.image;
            document.getElementById('modal-reporter-name').textContent = report.reporter;
            document.getElementById('modal-report-area').textContent = report.area;
            document.getElementById('modal-report-desc').textContent = report.description;

            const statusBar = document.getElementById('modal-status-bar');
            if (report.status === 'Pending') statusBar.style.width = '20%';
            else if (report.status === 'Under Review') statusBar.style.width = '45%';
            else if (report.status === 'In Progress') statusBar.style.width = '75%';
            else if (report.status === 'Resolved') statusBar.style.width = '100%';

            const repliesList = document.getElementById('modal-replies-list');
            if (report.replies.length === 0) {
                repliesList.innerHTML = `<p class="text-[11px] text-gray-400 italic">No official updates posted yet.</p>`;
            } else {
                repliesList.innerHTML = report.replies.map(rep => `
                    <div class="bg-emerald-50 dark:bg-brand-900 p-2.5 rounded-xl border border-emerald-100 dark:border-emerald-800">
                        <div class="flex justify-between items-center text-[10px] font-bold text-brand-700 dark:text-emerald-300">
                            <span><i class="fa-solid fa-circle-check"></i> ${rep.author}</span>
                            <span class="text-gray-400">${rep.date}</span>
                        </div>
                        <p class="text-xs text-gray-700 dark:text-gray-200 mt-1">${rep.text}</p>
                    </div>
                `).join('');
            }

            const adminBox = document.getElementById('admin-controls-box');
            if (state.isAdmin) {
                adminBox.classList.remove('hidden');
                document.getElementById('admin-status-select').value = report.status;
            } else {
                adminBox.classList.add('hidden');
            }

            document.getElementById('report-detail-modal').classList.remove('hidden');
        }

        function closeReportModal() {
            document.getElementById('report-detail-modal').classList.add('hidden');
        }

        function saveAdminResponse() {
            playUiSound('success');
            const report = state.reports.find(r => r.id === state.selectedReportId);
            if (!report) return;

            const newStatus = document.getElementById('admin-status-select').value;
            const replyText = document.getElementById('admin-reply-input').value;

            report.status = newStatus;
            if (replyText.trim()) {
                report.replies.push({
                    author: 'Clean Mohalla Official',
                    text: replyText,
                    date: new Date().toISOString().split('T')[0]
                });
            }

            showToast('Complaint Status Updated!');
            closeReportModal();
            renderCommunityFeed();
            renderHomeFeedPreview();
        }

            playUiSound('click');
